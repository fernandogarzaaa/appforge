┌─ COMPONENT QUICK REFERENCE ────────────────────────────────────────────────┐

SpectrumNavigation.jsx
  └─ Main layout container, wraps all pages
  └─ Props: { children, onSearchOpen }
  └─ Auto-includes: TopNav, Sidebar, Breadcrumbs, MobileDrawer

TopNav.jsx
  └─ Global header with search, notifications, user menu
  └─ Features: ⌘K search, 🌙 dark mode, 👤 user dropdown, 👑 admin badge
  └─ Props: { onMenuClick, onSearchOpen }

SpectrumSidebar.jsx
  └─ Collapsible left sidebar (desktop only)
  └─ Modes: Beginner, Advanced, Admin
  └─ Features: Smooth animation, dark mode, active highlighting
  └─ No props needed (uses useNavigation hook)

MobileDrawer.jsx
  └─ Mobile navigation overlay
  └─ Features: Slide animation, auto-close, touch-optimized
  └─ Props: { isOpen, onClose }

Breadcrumbs.jsx
  └─ Contextual navigation path
  └─ Auto-generates from route
  └─ Props: { customBreadcrumbs } (optional)

┌─ CONTEXT & HOOKS ──────────────────────────────────────────────────────────┐

useNavigation()
  └─ Returns:
     ├─ userMode: 'beginner' | 'advanced'
     ├─ isAdmin: boolean
     ├─ sidebarCollapsed: boolean
     ├─ darkMode: boolean
     ├─ currentRoute: string
     ├─ toggleSidebar(): void
     ├─ toggleDarkMode(): void
     ├─ updateUserMode(mode): void
     ├─ setAdminStatus(status): void
     └─ updateCurrentRoute(route): void

NavigationContext
  └─ Provider component (auto-wrapped by App.jsx)
  └─ Don't import directly, use useNavigation hook instead

┌─ ROUTE UTILITIES ──────────────────────────────────────────────────────────┐

getVisibleRoutes(userMode, isAdmin)
  └─ Returns routes visible to user
  └─ Example: getVisibleRoutes('advanced', false)

getRoutesByCategory(userMode, isAdmin)
  └─ Returns { main: [], advanced: [], admin: [] }
  └─ Useful for rendering grouped navigation

findRoute(path)
  └─ Returns single route object by path
  └─ Example: findRoute('/dashboard')

getBreadcrumbs(path)
  └─ Returns breadcrumb array for path
  └─ Example: getBreadcrumbs('/admin/users')

┌─ COMMON USAGE PATTERNS ────────────────────────────────────────────────────┐

// 1. USE NAVIGATION HOOK IN COMPONENT
import { useNavigation } from '@/hooks/useNavigation';

function MyComponent() {
  const { userMode, isAdmin, darkMode, toggleDarkMode } = useNavigation();
  
  return <div>Mode: {userMode}, Admin: {isAdmin}</div>;
}

// 2. CHECK ADMIN STATUS
function AdminOnlyFeature() {
  const { isAdmin } = useNavigation();
  
  if (!isAdmin) return <div>Not authorized</div>;
  return <AdminDashboard />;
}

// 3. CHECK USER MODE
function AdvancedFeature() {
  const { userMode } = useNavigation();
  
  if (userMode !== 'advanced') {
    return <div>Upgrade to advanced mode</div>;
  }
  return <AdvancedTool />;
}

// 4. GET VISIBLE ROUTES
import { getRoutesByCategory } from '@/lib/navigationRoutes';

const navRoutes = getRoutesByCategory('advanced', false);
// Use navRoutes.main, navRoutes.advanced, navRoutes.admin

// 5. CUSTOM BREADCRUMBS
import Breadcrumbs from '@/components/layout/Breadcrumbs';

<Breadcrumbs customBreadcrumbs={[
  { label: 'Dashboard', path: '/dashboard' },
  { label: 'My Page', path: '/mypage' }
]} />

┌─ RESPONSIVE BREAKPOINTS ───────────────────────────────────────────────────┐

Mobile (< 768px)
  └─ Sidebar: Hidden
  └─ Menu: Hamburger trigger
  └─ Drawer: Overlay on hamburger click
  └─ Search: Icon trigger

Tablet (768px - 1023px)
  └─ Sidebar: Visible (compact)
  └─ Menu: Hamburger visible
  └─ Drawer: Overlay available
  └─ Search: Full search bar

Desktop (≥ 1024px)
  └─ Sidebar: Always visible (280px expanded, 80px collapsed)
  └─ Menu: No hamburger
  └─ Drawer: Hidden
  └─ Search: Full search bar

┌─ TROUBLESHOOTING ──────────────────────────────────────────────────────────┐

Hook not working?
  └─ Ensure component is wrapped in Layout
  └─ Check App.jsx has NavigationProvider

Routes not showing?
  └─ Check visibility rules in navigationRoutes.js
  └─ Verify userMode and isAdmin values

Dark mode not applying?
  └─ Check localStorage['darkMode']
  └─ Verify Tailwind dark: prefix configured

Sidebar animation janky?
  └─ Check Framer Motion installed
  └─ Disable other animations

Admin items not showing?
  └─ Call setAdminStatus(true) after login
  └─ Check user object has isAdmin flag
